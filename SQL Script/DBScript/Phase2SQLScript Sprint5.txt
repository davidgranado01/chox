CREATE TABLE system_log
(
  id serial NOT NULL,
  action_id character varying(10),
  message character varying(100),
  status character varying(1),
  created_by integer,
  last_modified_by integer,
  last_modified_date timestamp without time zone,
  created_date timestamp without time zone,
  CONSTRAINT system_log_pkey PRIMARY KEY (id),
  CONSTRAINT action_fkey FOREIGN KEY (action_id)
      REFERENCES "action" (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (OIDS=FALSE);
ALTER TABLE system_log OWNER TO postgres;

create view vw_system_log as
select system_log.id, system_log.action_id, action.description, system_log.message, system_log.status, 
system_log.created_by, web_user.first_name||', '||web_user.last_name as name, web_user.email, system_log.created_date
from system_log system_log, action action, web_user web_user 
where action.id=system_log.action_id 
and system_log.created_by=web_user.id
order by created_date asc;

CREATE OR REPLACE VIEW rpt_claim_audit_trail AS 
 SELECT 
	audit_trail.id, 
	audit_trail.update_date, 
	audit_trail.user_id, 
	audit_trail.claim_id, 
	audit_trail.original_status, audit_trail.new_status, audit_trail.created_by, 
	audit_trail.created_date, audit_trail.last_modified_by, audit_trail.last_modified_date, 
	claim.insurer_id, claim.chorganisation_id, invoice.total_to_pay, 
	invoice.panalty_charge, invoice.id AS invoice_id, 
        CASE
            WHEN audit_trail.claim_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.claim_reason_of_rejection
        END AS claim_reason_of_rejection, 
        CASE
            WHEN audit_trail.invoice_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.invoice_reason_of_rejection
        END AS invoice_reason_of_rejection
   FROM audit_trail audit_trail
   JOIN claim claim ON claim.id = audit_trail.claim_id
   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id;

ALTER TABLE rpt_claim_audit_trail OWNER TO postgres;
