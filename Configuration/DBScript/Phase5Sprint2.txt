CREATE TABLE uploaded_claims_detail
(
  id serial NOT NULL,
  "version" integer NOT NULL,
  choReference character varying(128),
  is_valid boolean default true,
  message character varying(256),
  claim_status character varying(128),
  process_status character varying(128),
  remark character varying(128), 
  created_by integer NOT NULL,
  claim_id integer ,
  bordereau_id integer NOT NULL,
  created_date timestamp without time zone NOT NULL DEFAULT now(),
  last_modified_by integer NOT NULL,
  last_modified_date timestamp without time zone DEFAULT now(),
  CONSTRAINT uploaded_claims_detail_pkey PRIMARY KEY (id),
  CONSTRAINT created_by_fkey FOREIGN KEY (created_by)
      REFERENCES web_user (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION,
  CONSTRAINT last_modified_by_fkey FOREIGN KEY (last_modified_by)
      REFERENCES web_user (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION
  )
WITH (
  OIDS=FALSE
);

GRANT ALL ON TABLE uploaded_claims_detail TO chox;
GRANT SELECT, UPDATE, INSERT  ON TABLE uploaded_claims_detail TO chox_user;
GRANT SELECT, UPDATE, USAGE ON SEQUENCE uploaded_claims_detail_id_seq TO chox_user;
