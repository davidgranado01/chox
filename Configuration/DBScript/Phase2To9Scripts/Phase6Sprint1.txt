CREATE TABLE insurer_discount (
id serial NOT NULL,
  insurer_id integer NOT NULL,
  chorganisation_id integer NOT NULL,
  date_from timestamp without time zone NOT NULL,
  date_to timestamp without time zone NOT NULL,
  discount_percentage numeric(10,2),
  created_by integer NOT NULL,
  created_date timestamp without time zone NOT NULL DEFAULT now(),
  last_modified_by integer NOT NULL,
  last_modified_date timestamp without time zone DEFAULT now(),
  CONSTRAINT insurer_discount_pkey PRIMARY KEY (id),
  CONSTRAINT insurer_discount_ref_cho_fkey FOREIGN KEY (chorganisation_id)
      REFERENCES chorganisation (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION,
  CONSTRAINT insurer_discount_ref_ins_fkey FOREIGN KEY (insurer_id)
      REFERENCES insurer (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION,
  CONSTRAINT billing_cho_ref_webuser_created_fkey FOREIGN KEY (created_by)
      REFERENCES web_user (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION,
  CONSTRAINT billing_cho_ref_webuser_modified_fkey FOREIGN KEY (last_modified_by)
      REFERENCES web_user (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (
  OIDS=FALSE
);
GRANT SELECT, UPDATE, INSERT, DELETE ON TABLE insurer_discount TO chox_user;
GRANT SELECT, UPDATE, USAGE ON SEQUENCE insurer_discount_id_seq TO chox_user;


ALTER TABLE invoice ADD COLUMN insurer_discount numeric(10,2) NOT NULL DEFAULT 0.00;

ALTER TABLE invoice_original ADD COLUMN insurer_discount numeric(10,2);

ALTER TABLE comment ADD COLUMN  raised_by integer;

ALTER TABLE insurer ADD COLUMN is_insurer_discount_enable boolean NOT NULL DEFAULT false;
