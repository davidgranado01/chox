--- 6.11.2 Move 'Enable Repair Documentation Automated Tasks' to BRE Band
--------------------------------------------------------------------------------------
ALTER TABLE bre_band ADD COLUMN allow_manag_repair_auto_tasks boolean not null default false;
ALTER TABLE bre_band ADD COLUMN allow_notmanag_repair_auto_tasks boolean not null default false;

UPDATE bre_band
  set allow_manag_repair_auto_tasks = true,
      allow_notmanag_repair_auto_tasks = true
from bre_band_organisation bbo, chorganisation c
where bbo.band_id = bre_band.id
  and bbo.chorganisation_id = c.id
  and c.is_allow_repairdoc_automated_tasks = true;

ALTER TABLE chorganisation DROP COLUMN is_allow_repairdoc_automated_tasks;

--- bug#1755: Production - CHO MI Access Role Can See 'Switch Claim' Button
                          When They Should Not Be Able To
--------------------------------------------------------------------------------------
insert into accessibility_item(accessibility_id, role, access_right)
select accessibility_id, 'ROLE_CHO_OPR', 1 from accessibility_item where accessibility_id in (select id from accessibility where name ilike '%switch%') and role='ROLE_CHO' and access_right!=0;

insert into accessibility_item(accessibility_id, role, access_right)
select accessibility_id, 'ROLE_CHO_MNG', 1 from accessibility_item where accessibility_id in (select id from accessibility where name ilike '%switch%') and role='ROLE_CHO' and access_right!=0;

delete from accessibility_item
using accessibility
where accessibility_item. accessibility_id = accessibility.id
  and accessibility.name ilike '%switch%'
  and accessibility_item.role='ROLE_CHO'
  and accessibility_item.access_right=1;
