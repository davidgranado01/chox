CREATE OR REPLACE FUNCTION updateinvoiceoriginal()
  RETURNS SETOF claim AS
$BODY$

DECLARE

claimRecord claim%rowtype;

BEGIN

RAISE NOTICE 'invoice original update process started' ;

  FOR claimRecord IN 
   SELECT * 
   FROM
     claim c,
     invoice i
   WHERE 
     c.invoice_id = i.id
     AND c.invoice_original_id is null 

  LOOP

    -- RAISE NOTICE 'Cho Reference : % ',claimRecord.cho_reference;
    
    INSERT INTO invoice_original 
     (
      date_invoiced, miscellaneous_qty, automatic_qty, sat_nav_qty, estate_qty, baby_seat_qty, 
      tow_bars_qty, non_standard_insurance_premium_qty, admin_qty, roof_rack_qty, dual_control_qty,
      delivery_collection_qty, created_by, created_date, last_modified_by, last_modified_date, hire_net, 
      hire_vat, hire_gross, repair_net, repair_vat, repair_gross, engineer_fee_net, engineer_fee_vat, engineer_fee_gross,
      storage_recovery_net, storage_recovery_vat, storage_recovery_gross, total_net, total_vat, total_gross,
      claims_handling_invoice_amount, deduction_for_claims_handling_fee, discount, full_total_to_pay, original_full_total_to_pay, miscellaneous_fee,
      automatic_fee, sat_nav_fee, estate_fee, baby_seat_fee, tow_bars_fee, non_standard_insurance_premium_fee, 
      admin_fee, roof_rack_fee, dual_control_fee, delivery_collection_fee, 
      hire_rate_charged_per_day, excess_amount_collected, 
      vat_amount_collected, version, total_to_pay, original_total_to_pay,
      additional_driver_fee, additional_driver_qty, total_loss_net, total_loss_vat, total_loss_gross,
      insurer_discount
     ) 
       SELECT 
        i.date_invoiced, i.miscellaneous_qty, i.automatic_qty, i.sat_nav_qty, i.estate_qty, i.baby_seat_qty,
        i.tow_bars_qty, i.non_standard_insurance_premium_qty, i.admin_qty, i.roof_rack_qty, i.dual_control_qty,
        i.delivery_collection_qty, 999, now(), 999, now(), i.hire_net, 
        i.hire_vat, i.hire_gross, i.repair_net, i.repair_vat, i.repair_gross, i.engineer_fee_net, i.engineer_fee_vat, i.engineer_fee_gross,
        i.storage_recovery_net, i.storage_recovery_vat, i.storage_recovery_gross, i.total_net, i.total_vat, i.total_gross, 
        i.claims_handling_invoice_amount, i.deduction_for_claims_handling_fee, i.discount, i.full_total_to_pay, i.original_full_total_to_pay, i.miscellaneous_fee,
        i.automatic_fee, i.sat_nav_fee, i.estate_fee, i.baby_seat_fee, i.tow_bars_fee, i.non_standard_insurance_premium_fee, 
        i.admin_fee, i.roof_rack_fee, i.dual_control_fee, i.delivery_collection_fee, 
        i.hire_rate_charged_per_day, i.excess_amount_collected, 
        i.vat_amount_collected, 0, i.total_to_pay, i.original_total_to_pay,
        i.additional_driver_fee, i.additional_driver_qty, i.total_loss_net, i.total_loss_vat, i.total_loss_gross,
        i.insurer_discount
       FROM invoice i 
       WHERE i.id = claimRecord.invoice_id;
       
    UPDATE claim SET invoice_original_id = (SELECT currval(pg_get_serial_sequence('invoice_original', 'id'))), version = claim.version + 1, last_modified_date = now(), last_modified_by = 999 WHERE id = claimRecord.id;

    RETURN NEXT claimRecord;
  END LOOP;
RAISE NOTICE 'invoice original update process finished' ;
RAISE NOTICE 'Updating total_to_pay column';

UPDATE invoice_original SET total_to_pay = original_total_to_pay, version = invoice_original.version + 1, last_modified_date = now(), last_modified_by = 999 WHERE total_to_pay != original_total_to_pay AND original_total_to_pay > 0;

RAISE NOTICE 'Updating full_total_to_pay column';

UPDATE invoice_original SET full_total_to_pay = original_full_total_to_pay, version = invoice_original.version + 1, last_modified_date = now(), last_modified_by = 999 WHERE full_total_to_pay != original_full_total_to_pay AND original_full_total_to_pay > 0;


RETURN ;
END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100
  ROWS 1000;

CREATE OR REPLACE FUNCTION bre_failure_report(IN startdate text, IN enddate text, IN insurerid integer)
  RETURNS TABLE(ruleid character varying, rulename character varying, numberoffailures bigint, percentage numeric, numberofuniquecases bigint, averageinvoicedvalue numeric, averagessettledvalue numeric) AS
$BODY$
DECLARE
BEGIN 

RETURN QUERY 
select h.rule_id as RuleId, n.name,
count(*) as NumberOfFailures, (100.0*count(*)/(select count(*)
      from claim, invoice, history
      where  claim.invoice_id = invoice.id
             and claim.id = history.claim_id
             and  invoice.created_date between startDate::date and endDate::date -- invoice created date
             and claim.insurer_id in (insurerId) -- insurer ids
             and history.claim_id = claim.id  
             and history.type = 'ERROR'))::numeric(6,3) as Percentage,

      count(distinct(c.id)) as NumberOfUniqueCases,

      (select avg(invoice_original.total_to_pay)::numeric(8,2) 
       from invoice_original, claim
       inner join invoice on claim.invoice_id = invoice.id
       where invoice.invoice_original_id = invoice_original.id and claim.id in (select distinct(c.id)
from history hi, invoice i, claim c 
where c.invoice_id = i.id
and c.id = hi.claim_id
and  i.created_date between startDate::date and endDate::date -- invoice created date
and c.insurer_id in (insurerId) -- insurer ids
and hi.type = 'ERROR' 
and hi.rule_id = h.rule_id)) as AveragesSettledValue,

     (select avg(inv.total_to_pay)::numeric(8,2) 
     from invoice inv 
     where inv.id in (select distinct(i.id) 
from history hi, invoice i, claim c 
where c.invoice_id = i.id
and c.id = hi.claim_id
and i.created_date between startDate::date and endDate::date -- invoice created date
and c.insurer_id in (insurerId) -- insurer ids
and hi.type = 'ERROR'
and hi.rule_id = h.rule_id)) as AverageInvoicedValue

from rule_name n, history h, invoice i, invoice_original io, claim c 
where n.rule_id = h.rule_id
   and i.invoice_original_id = io.id
   and c.invoice_id = i.id
   and c.id = h.claim_id
   and  i.created_date between startDate::date and endDate::date -- invoice created date
   and c.insurer_id in (insurerId) -- insurer ids
   and h.type = 'ERROR' 
group by h.rule_id, n.name order by h.rule_id;END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100
  ROWS 1000;

create or replace function monthly_mobile_chox_report()
returns table("Month" text, No_Notifications_Received bigint, No_Invoices_Received bigint, No_Invoices_Paid bigint,
           Value_of_Invoices_Received numeric(15,2), Value_of_Invoices_Paid numeric(15,2), Average_Days_Invoice_Loaded_to_Payment numeric(8,2), Cumulative_Total_Invoices_Submitted bigint,
           Cumulative_Total_Invoices_Paid bigint, Cumulative_Total_Invoices_Outstanding bigint, No_of_CHOS_Onboard bigint, No_Manual_CHOs bigint, No_of_Insurers_Onboard bigint, No_of_CHO_Insurer_Pairings bigint, Number_of_registered_users bigint) as $$

DECLARE
   first_claim_date date;
   start_date date;
   end_date date;
BEGIN

   first_claim_date = (select min(created_date) FROM claim);
   start_date = (select to_date(to_char(first_claim_date, 'MM') || '-01-' || to_char(first_claim_date, 'yyyy'), 'mm-dd-yyyy'));
   end_date = start_date +  interval '1 month';

WHILE start_date < now() LOOP

   
RETURN QUERY

   select to_char(start_date, 'TMMonth') || ' - ' || to_char(start_date , 'yyyy') AS "Month",

  (select count(*) as No_Notifications_Received from claim where created_date between start_date and end_date),
       
  (select count(*) as No_Invoices_Received from invoice where created_date between start_date and end_date),

  (select count(*) as No_Invoices_Paid from audit_trail a where a.new_status='InvoicePaymentLogged' 
       and a.update_date between start_date and end_date
       and a.reverted=false),
                  
  (select sum(io.total_to_pay)::numeric(15,2) as Value_of_Invoices_Received 
       from invoice i, invoice_original io, claim c
       where c.invoice_id = i.id and i.invoice_original_id = io.id and i.created_date between start_date and end_date),
       
  (select sum(i.total_to_pay)::numeric(15,2) as Value_of_Invoices_Paid from audit_trail a, invoice i, claim c
       where a.new_status='InvoicePaymentLogged' and a.update_date between start_date and end_date
       and a.reverted=false
       and c.id = a.claim_id and c.invoice_id = i.id),
                      
   (select case when count(*) = 0 then 0 else avg(EXTRACT(DAY FROM (a.update_date - i.created_date)))::numeric(8,2) end as Average_Days_Invoice_Loaded_to_Payment 
              from invoice i, claim c, audit_trail a where a.claim_id = c.id and c.invoice_id = i.id and a.new_status in ('InvoicePaymentLogged', 'ManualInvoicePaid') 
              and a.reverted=false and i.created_date between start_date and end_date),
              
   (select count(*) as Cumulative_Total_Invoices_Submitted from invoice where created_date < end_date),
   
   (select count(*) as Cumulative_Total_Invoices_Paid from audit_trail a where a.new_status='InvoicePaymentLogged' and a.update_date < end_date and a.reverted=false),

   (select count(*) as Cumulative_Total_Invoices_Outstanding
        from claim c , audit_trail a
        where a.claim_id = c.id
              and a.new_status in ('InvoiceReferredToEngineer','InvoiceEscalated', 'InvoiceDataCalculationIncorrect', 'InvoiceApprovedByBRE', 'ContestedInvoiceReferredToInsurer',
                               'ContestedInvoiceReferredToCHO', 'InvoiceReferredToClaimsHandler', 'AwaitingLiabilityResolution', 'InvoiceUnassigned', 'InvoiceEscalatedToHandler', 
                               'ManualInvoiceBRERejected', 'ManualInvoiceContested', 'ManualInvoiceBREApproved', 'AwaitingLitigationOutcome')
              and a.created_date < end_date
              and (a.reverted = false or (a.reverted=true and a.last_modified_date > end_date))  -- check it wasn't reverted, or if it was, it was reverted AFTER the end of the month
              and not exists (select * from audit_trail a2 where a2.claim_id = c.id and a2.created_date > a.created_date and a2.created_date < end_date)),
              
   (select count(*) as No_of_CHOS_Onboard from chorganisation where status = true and insurer_upload_only = false and created_date < end_date),
   
   (select count(*) as No_Manual_CHOs from chorganisation where status = true and insurer_upload_only = true and created_date < end_date),
              
   (select count(*) as No_of_Insurers_Onboard from insurer where status = true and created_date < end_date),

   (select count(*) as No_of_CHO_Insurer_Pairings from insurer_chorganisation ic , insurer i, chorganisation c where ic.insurer_id = i.id 
                   and ic.chorganisation_id = c.id and c.status = true and i.status = true and ic.status = true and ic.created_date < end_date),
              
   (select count(distinct wu.id) from web_user wu, web_user_user_role wuur, web_user_role wur  where wu.id = wuur.web_user_id 
              and wur.name != 'ROLE_CHOX_ADMIN' 
              and wuur.web_user_role_id = wur.id 
              and wu.status = true and wu.created_date < end_date);


start_date = end_date;
end_date = end_date + interval '1 month';

END LOOP;
END;
$$ LANGUAGE plpgsql;

repairpenalPer30DVal NUMERIC(8,3);
repairpenalPer60D NUMERIC;
repairpenalPer60DVal NUMERIC(8,3);

BEGIN
ClaimsId = claimId;
userId = userIdNumber;
hirepenalPer30D = 7.5;
hirepenalPer30DVal = hirepenalPer30D/100;
hirepenalPer60D = 15;
hirepenalPer60DVal = hirepenalPer60D/100;
repairpenalPer30D = 2.5;

invoiceCount90D = 0;

FOR claimRecord30D IN

FROM
 claim c,
 invoice i, 
 bre_band bre, 
 bre_band_organisation breorg, 
 insurer ins, 
 chorganisation cho

 c.invoice_id = i.id 
 AND (c.id = ClaimsId or ClaimsId = -1)

 AND c.chorganisation_id = cho.id
 AND bre.insurer_id = ins.id
 AND breorg.chorganisation_id = cho.id
 AND breorg.band_id = bre.id
 AND bre.allow_penalty_charges = true
 AND c.auto_penalty_charges = true
 AND cho.auto_penalty_charges = true
 AND i.penalty_alert_qty = 0
 AND c.status NOT IN ( 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoicePaymentLogged', 'InvoiceDataCalculationIncorrect')
 AND (current_date - i.auto_penalty_start::Date) >= 30
 AND (current_date - i.auto_penalty_start::Date) < 60

LOOP


RAISE NOTICE 'Cho Reference for invoice > 30 days : % ',claimRecord30D.cho_reference;
    
    UPDATE invoice  
    SET 
      auto_penalty_alert_qty = 1,
      penalty_alert_qty = 1,
      hire_penalty_charge_applied_date = (CASE WHEN(hire_net > 0) THEN now() ELSE null END),
      repair_penalty_charge_applied_date = (CASE WHEN(repair_net > 0) THEN now() ELSE null END),
      full_total_to_pay = (full_total_to_pay - (hire_penalty_charge + repair_penalty_charge) + (hire_gross * hirepenalPer30DVal) + (repair_gross * repairpenalPer30DVal))::NUMERIC(8,2),
      total_to_pay = (CASE WHEN (claimRecord30D.liability_status = 5 OR claimRecord30D.liability_status = 6)

                           ELSE (full_total_to_pay - (hire_penalty_charge + repair_penalty_charge) + (hire_gross * hirepenalPer30DVal) + (repair_gross * repairpenalPer30DVal))::NUMERIC(8,2) 
                           END ),
      hire_penalty_charge = (hire_gross * hirepenalPer30DVal)::NUMERIC(8,2),
      repair_penalty_charge = (repair_gross * repairpenalPer30DVal)::NUMERIC(8,2),
      hire_penalty_percentage = (CASE WHEN(hire_net > 0) THEN hirepenalPer30D || '%' ELSE null END),
      repair_penalty_percentage = (CASE WHEN(repair_net > 0) THEN repairpenalPer30D || '%' ELSE null END),
      total_penalty_charge =  ((hire_gross * hirepenalPer30DVal) + (repair_gross * repairpenalPer30DVal))::NUMERIC(8,2),
      last_modified_by = userId,
      last_modified_date = now(),
      version = version + 1
   WHERE
      claimRecord30D.invoice_id = invoice.id;

--RAISE NOTICE 'Invoice > 30 days hirepenalPer30DVal :% , repairpenalPer30DVal :% ', hirepenalPer30DVal , repairpenalPer30DVal;
--RAISE NOTICE 'Invoice > 30 days hire gross :%  ', (select i.hire_gross   from invoice i where i.id = claimRecord30D.invoice_id);
--RAISE NOTICE 'Invoice > 30 days repair gross :%  ', (select i.repair_gross from invoice i where i.id = claimRecord30D.invoice_id);
--RAISE NOTICE 'Invoice > 30 hire penalty charge  :%  ', (select i.hire_gross from invoice i where i.id = claimRecord30D.invoice_id) * hirepenalPer30DVal;
--RAISE NOTICE 'Invoice > 30 repair penalty charge  :%  ', (select i.repair_gross from invoice i where i.id = claimRecord30D.invoice_id) * hirepenalPer30DVal;


--RAISE NOTICE 'Invoice > 30 days updated ';

   INSERT INTO comment
      (claim_id,created_by,created_date,last_modified_by,last_modified_date,visibility_type,comment)  
   SELECT 
      claimRecord30D.id,
      userId,
      now() + interval '0.1 sec',
      userId,
      now() + interval '0.1 sec',
      0,
      'Automatic penalty charges of £' ||  ((i.hire_gross * hirepenalPer30DVal) + (i.repair_gross * repairpenalPer30DVal))::NUMERIC(10,2) 
                                      || ' have been applied to the invoice as the age of the invoice has exceeded ' 
                                      || (current_date - i.auto_penalty_start::Date) 
                                      || ' days.'
   FROM

   WHERE
