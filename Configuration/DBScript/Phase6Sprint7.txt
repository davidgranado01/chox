-- Add a ClaimType field to Claim:
--   this replaces the following flags: is_tpi_claim, is_insurer_vs_insurer_claim, original_supp_inv, supplementary_invoiced_claim
ALTER TABLE claim ADD COLUMN claim_type integer not null default 0;

UPDATE claim SET claim_type = 1   -- GTA_ORIGINAL_INVOICE
where is_tpi_claim = false
  and is_insurer_vs_insurer_claim=false
  and original_supp_inv = true;

UPDATE claim SET claim_type = 2   -- GTA_SUPPLEMENTARY_INVOICE
where is_tpi_claim = false
  and is_insurer_vs_insurer_claim=false
  and supplementary_invoiced_claim=true
  and original_supp_inv = false;

UPDATE claim SET claim_type = 3   -- TPI
where is_tpi_claim = true
  and supplementary_invoiced_claim=false
  and original_supp_inv = false;

UPDATE claim SET claim_type = 4   -- INSURER_VS_INSURER
where is_insurer_vs_insurer_claim=true
  and supplementary_invoiced_claim=false
  and original_supp_inv = false;

UPDATE claim SET claim_type = 5   -- INSURER_VS_INSURER_ORIGINAL_INVOICE
where is_insurer_vs_insurer_claim=true
  and original_supp_inv = true;

UPDATE claim SET claim_type = 6   -- INSURER_VS_INSURER_SUPPLEMENTARY_INVOICE
where is_insurer_vs_insurer_claim=true
  and supplementary_invoiced_claim=true
  and original_supp_inv = false;

UPDATE claim SET claim_type = 10   -- INSURER_UPLOAD
where insurer_upload = true;

DROP VIEW rpt_claim_audit_trail;

-- View: rpt_claim_audit_trail

CREATE OR REPLACE VIEW rpt_claim_audit_trail AS 
 SELECT audit_trail.id, audit_trail.update_date, audit_trail.user_id, audit_trail.claim_id, audit_trail.original_status, audit_trail.new_status, audit_trail.reverted, audit_trail.created_by, audit_trail.created_date, audit_trail.last_modified_by, audit_trail.last_modified_date, claim.insurer_id, claim.chorganisation_id, claim.claim_type, invoice.full_total_to_pay AS total_to_pay, invoice.hire_penalty_charge AS penalty_charge, invoice.id AS invoice_id, 
        CASE
            WHEN audit_trail.claim_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.claim_reason_of_rejection
        END AS claim_reason_of_rejection, 
        CASE
            WHEN audit_trail.invoice_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.invoice_reason_of_rejection
        END AS invoice_reason_of_rejection
   FROM audit_trail audit_trail
   JOIN claim claim ON claim.id = audit_trail.claim_id
   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id;

-- Function: addinvoicepenaltytask(integer)
--
DROP FUNCTION addinvoicepenaltytask(integer);

CREATE OR REPLACE FUNCTION addinvoicepenaltytask(integer)
  RETURNS boolean AS
$BODY$

DECLARE
userId int;

BEGIN

userId=$1;

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '15 days', 'Invoice Approaching 90 Days', 'The invoice was uploaded over 75 days ago and may be subject to penalties in 15 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty = 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.created_date)/(3600*24.0) >= 75
  and extract(epoch from now() - i.created_date)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '10 days')
  and ((liability_status is null or (liability_status !=5 and liability_status!=6)) or ((liability_status = 5 or liability_status =6 ) and extract(epoch from now() - c.liability_agreed_date)/(3600*24) >= 75));

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '5 days', 'Invoice Approaching ' || 30*(i.penalty_alert_qty+1) || ' Days', 'The invoice was uploaded over ' || 30*(i.penalty_alert_qty+1)-5 || ' days ago and may be subject to penalties in 5 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty >= 0 and i.penalty_alert_qty < 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.created_date)/(3600*24.0) >= ((i.penalty_alert_qty+1)*30 - 5)
  and extract(epoch from now() - i.created_date)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '29 days')
  and ((liability_status is null or (liability_status !=5 and liability_status!=6)) or ((liability_status = 5 or liability_status =6 ) and extract(epoch from now() - c.liability_agreed_date)/(3600*24) > ((i.penalty_alert_qty+1)*30)-5));


update task set complete = true,
                completed_by = 999,
                completed_date = now()
where due_date < now() and task_type like 'Invoice Approaching%' and complete = false;


return true;

END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100;

GRANT EXECUTE ON FUNCTION addinvoicepenaltytask(integer) TO public;
GRANT EXECUTE ON FUNCTION addinvoicepenaltytask(integer) TO chox_user;

$BODY$

BEGIN

userId = userIdNumber;
hirepenalPer30D = 7.5;
hirepenalPer30DVal = hirepenalPer30D/100;

invoiceCount30D = 0;
invoiceCount60D = 0;
invoiceCount90D = 0;

RAISE NOTICE 'userid : % , hirepenalPer30DVal : %,  hirepenalPer60DVal: %,  repairpenalPer30DVal: %, repairpenalPer60DVal : %',userId,hirepenalPer30DVal,hirepenalPer60DVal,repairpenalPer30DVal,repairpenalPer60DVal;


FOR claimRecord30D IN 
SELECT * 
FROM
 claim c,

--
-- Alter Store procedure to create automatic task
--
CREATE OR REPLACE FUNCTION addinvoicepenaltytask(integer)
  RETURNS boolean AS
$BODY$


DECLARE
userId int;

BEGIN

userId=$1;

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '15 days', 'Invoice Approaching 90 Days', 'The invoice was uploaded over 75 days ago and may be subject to penalties in 15 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty <= 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived')
  and (current_date - i.auto_penalty_start::Date) >= 75
  and (current_date - i.auto_penalty_start::Date) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '10 days');

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '5 days', 'Invoice Approaching ' || 30*(i.penalty_alert_qty+1) || ' Days', 'The invoice was uploaded over ' || 30*(i.penalty_alert_qty+1)-5 || ' days ago and may be subject to penalties in 5 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty >= 0 and i.penalty_alert_qty < 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived')
  and (current_date - i.auto_penalty_start::Date) >= ((i.penalty_alert_qty+1)*30 - 5)
  and (current_date - i.auto_penalty_start::Date) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '29 days');


update task set complete = true,
                completed_by = userId,
                completed_date = now()
where due_date::Date < current_date and task_type like 'Invoice Approaching%' and complete = false;


return true;

END;
$BODY$
