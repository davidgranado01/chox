-- Add a ClaimType field to Claim:
--   this replaces the following flags: is_tpi_claim, is_insurer_vs_insurer_claim, original_supp_inv, supplementary_invoiced_claim
ALTER TABLE claim ADD COLUMN claim_type integer not null default 0;

UPDATE claim SET claim_type = 1   -- GTA_ORIGINAL_INVOICE
where is_tpi_claim = false
  and is_insurer_vs_insurer_claim=false
  and original_supp_inv = true;

UPDATE claim SET claim_type = 2   -- GTA_SUPPLEMENTARY_INVOICE
where is_tpi_claim = false
  and is_insurer_vs_insurer_claim=false
  and supplementary_invoiced_claim=true
  and original_supp_inv = false;

UPDATE claim SET claim_type = 3   -- TPI
where is_tpi_claim = true
  and supplementary_invoiced_claim=false
  and original_supp_inv = false;

UPDATE claim SET claim_type = 4   -- INSURER_VS_INSURER
where is_insurer_vs_insurer_claim=true
  and supplementary_invoiced_claim=false
  and original_supp_inv = false;

UPDATE claim SET claim_type = 5   -- INSURER_VS_INSURER_ORIGINAL_INVOICE
where is_insurer_vs_insurer_claim=true
  and original_supp_inv = true;

UPDATE claim SET claim_type = 6   -- INSURER_VS_INSURER_SUPPLEMENTARY_INVOICE
where is_insurer_vs_insurer_claim=true
  and supplementary_invoiced_claim=true
  and original_supp_inv = false;

UPDATE claim SET claim_type = 10   -- INSURER_UPLOAD
where insurer_upload = true;


ALTER TABLE CLAIM DROP COLUMN is_tpi_claim;
ALTER TABLE CLAIM DROP COLUMN is_insurer_vs_insurer_claim;
ALTER TABLE CLAIM DROP COLUMN original_supp_inv;
ALTER TABLE CLAIM DROP COLUMN supplementary_invoiced_claim;
ALTER TABLE CLAIM DROP COLUMN insurer_upload;

CREATE OR REPLACE VIEW rpt_claim_audit_trail AS 
 SELECT audit_trail.id, audit_trail.update_date, audit_trail.user_id, audit_trail.claim_id, audit_trail.original_status, audit_trail.new_status, audit_trail.created_by, audit_trail.created_date, audit_trail.last_modified_by, audit_trail.last_modified_date, claim.insurer_id, claim.chorganisation_id, claim.claim_type, invoice.full_total_to_pay AS total_to_pay, invoice.hire_penalty_charge AS penalty_charge, invoice.id AS invoice_id,

            WHEN audit_trail.claim_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.claim_reason_of_rejection
        END AS claim_reason_of_rejection, 
        CASE
            WHEN audit_trail.invoice_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.invoice_reason_of_rejection
        END AS invoice_reason_of_rejection
   FROM audit_trail audit_trail

   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id;

-- Function: addinvoicepenaltytask(integer)
--
DROP FUNCTION addinvoicepenaltytask(integer);

CREATE OR REPLACE FUNCTION addinvoicepenaltytask(integer)
  RETURNS boolean AS
$BODY$

DECLARE
userId int;

BEGIN

userId=$1;

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '15 days', 'Invoice Approaching 90 Days', 'The invoice was uploaded over 75 days ago and may be subject to penalties in 15 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty = 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.created_date)/(3600*24.0) >= 75
  and extract(epoch from now() - i.created_date)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '10 days')
  and ((liability_status is null or (liability_status !=5 and liability_status!=6)) or ((liability_status = 5 or liability_status =6 ) and extract(epoch from now() - c.liability_agreed_date)/(3600*24) >= 75));

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '5 days', 'Invoice Approaching ' || 30*(i.penalty_alert_qty+1) || ' Days', 'The invoice was uploaded over ' || 30*(i.penalty_alert_qty+1)-5 || ' days ago and may be subject to penalties in 5 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty >= 0 and i.penalty_alert_qty < 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.created_date)/(3600*24.0) >= ((i.penalty_alert_qty+1)*30 - 5)
  and extract(epoch from now() - i.created_date)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '29 days')
  and ((liability_status is null or (liability_status !=5 and liability_status!=6)) or ((liability_status = 5 or liability_status =6 ) and extract(epoch from now() - c.liability_agreed_date)/(3600*24) > ((i.penalty_alert_qty+1)*30)-5));


update task set complete = true,
                completed_by = 999,
                completed_date = now()
where due_date < now() and task_type like 'Invoice Approaching%' and complete = false;


return true;

END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100;
ALTER FUNCTION addinvoicepenaltytask(integer)
  OWNER TO chox;
GRANT EXECUTE ON FUNCTION addinvoicepenaltytask(integer) TO chox;
GRANT EXECUTE ON FUNCTION addinvoicepenaltytask(integer) TO public;
GRANT EXECUTE ON FUNCTION addinvoicepenaltytask(integer) TO chox_user;

CREATE OR REPLACE FUNCTION applyAutoPenaltyCharge(userIdNumber integer) 
	RETURNS boolean AS 
$BODY$

DECLARE
userId int;
vatRate NUMERIC(8,2);
vatRateVal NUMERIC(8,2);
hirepenalPer30D NUMERIC(8,2);
hirepenalPer30DVal NUMERIC(8,2);
hirepenalPer60D NUMERIC(8,2);
hirepenalPer60DVal NUMERIC(8,2);
repairpenalPer30D NUMERIC(8,2);
repairpenalPer30DVal NUMERIC(8,2);
repairpenalPer60D NUMERIC(8,2);
repairpenalPer60DVal NUMERIC(8,2);

BEGIN

userId = userIdNumber;
vatRate = 20;
vatRateVal = vatRate/100;
hirepenalPer30D = 7.5;
hirepenalPer30DVal = hirepenalPer30D/100;
hirepenalPer60D = 15;
hirepenalPer60DVal = hirepenalPer60D/100;
repairpenalPer30D = 2.5;
repairpenalPer30DVal = repairpenalPer30D/100;
repairpenalPer60D = 5;
repairpenalPer60DVal = repairpenalPer60D/100;

RAISE NOTICE 'userid : % , given vat rate : %, vatRateVal: % , hirepenalPer30DVal : %,  hirepenalPer60DVal: %,  repairpenalPer30DVal: %, repairpenalPer60DVal : %',userId,vatRate, vatRateVal,hirepenalPer30DVal,hirepenalPer60DVal,repairpenalPer30DVal,repairpenalPer60DVal;

INSERT INTO comment
(claim_id,created_by,created_date,last_modified_by,last_modified_date,visibility_type,comment)  
SELECT 
 c.id,
 userId,
 now(),
 userId,
 now(),
 0,
 'Automatic penalty charge of £' ||  (params.hirePenaltyCal + params.repairPenaltyCal)::NUMERIC(10,2) 
                                 || ' applied to the invoice as the age of the invoice exceeded ' 
                                 || (extract(epoch from now() - i.auto_penalty_start)/(3600*24.0))::int 
                                 || ' days.'
from
 claim c,
 invoice i, 
 bre_band bre , 
 bre_band_organisation breorg , 
 insurer ins , 
 chorganisation cho,
 (
  select 
   (i2.hire_net * hirepenalPer30DVal) + (i2.hire_net * hirepenalPer30DVal * subParams.vatUsed) as hirePenaltyCal,
   (i2.repair_net * repairpenalPer30DVal) + (i2.repair_net * repairpenalPer30DVal * subParams.vatUsed) as repairPenaltyCal,
   i2.id as invId
  from
   (
     select 
      COALESCE((i1.hire_vat/i1.hire_net),vatRateVal)::NUMERIC(6,2) as vatUsed,
      i1.id as iid 
     from 
      invoice i1
   ) subParams,
   invoice i2
  where 
   i2.id = subParams.iid
 ) params
where 
 params.invId = i.id
 and c.invoice_id = i.id 
 and c.claim_type not in (3,4,5,6,7,8,9,10)

 and cho.auto_penalty_charges = true
 and i.penalty_alert_qty = 0
 and c.status not in ( 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived','InvoicePaymentLogged','InvoiceDataCalculationIncorrect')
 and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) >= 30 
 and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) < 60;


RAISE NOTICE 'Notes has been created for invoice older than 30 days '; 


update invoice  
set

 hire_penalty_charge_applied_date = (CASE WHEN(hire_net > 0) THEN now() ELSE null END),
 repair_penalty_charge_applied_date = (CASE WHEN(repair_net > 0) THEN now() ELSE null END),
 full_total_to_pay = full_total_to_pay - (hire_penalty_charge + repair_penalty_charge) + params.hirePenaltyCal + params.repairPenaltyCal,
 total_to_pay = (CASE WHEN (c.liability_status = 5 OR c.liability_status = 6) 
                      THEN (c.percentage_liability_accepted/100) * (full_total_to_pay - (hire_penalty_charge + repair_penalty_charge) + params.hirePenaltyCal + params.repairPenaltyCal)
                      ELSE full_total_to_pay - (hire_penalty_charge + repair_penalty_charge) + params.hirePenaltyCal + params.repairPenaltyCal 
                      END ),
 hire_penalty_charge = params.hirePenaltyCal,
 repair_penalty_charge = params.repairPenaltyCal,
 hire_penalty_percentage = (CASE WHEN(hire_net > 0) THEN hirepenalPer30D || '%' ELSE null END),
 repair_penalty_percentage = (CASE WHEN(repair_net > 0) THEN repairpenalPer30D || '%' ELSE null END),
 total_penalty_charge =  params.hirePenaltyCal + params.repairPenaltyCal,
 last_modified_by = userId,
 last_modified_date = now()
from
 claim c, 
 bre_band bre , 
 bre_band_organisation breorg , 
 insurer ins , 
 chorganisation cho,
 (
  select 
   (i2.hire_net * hirepenalPer30DVal) + (i2.hire_net * hirepenalPer30DVal * subParams.vatUsed) as hirePenaltyCal,

     from

   ) subParams,
   invoice i2
  where

 ) params
where 
 params.invId = invoice.id
 and c.invoice_id = invoice.id 
 and c.claim_type not in (3,4,5,6,7,8,9,10)
 and c.insurer_id = ins.id 
 and c.chorganisation_id = cho.id
 and bre.insurer_id = ins.id
 and breorg.chorganisation_id = cho.id
 and breorg.band_id = bre.id
 and bre.allow_penalty_charges = true
 and c.auto_penalty_charges = true
 and cho.auto_penalty_charges = true
 and invoice.penalty_alert_qty = 0
 and c.status not in ( 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived','InvoicePaymentLogged','InvoiceDataCalculationIncorrect')
 and extract(epoch from now() - invoice.auto_penalty_start)/(3600*24.0) >= 30

RAISE NOTICE 'Penalty charge for invoice older than 30 days applied ';

--

CREATE OR REPLACE FUNCTION addinvoicepenaltytask(integer)
  RETURNS boolean AS
$BODY$


DECLARE
userId int;

BEGIN

userId=$1;

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '15 days', 'Invoice Approaching 90 Days', 'The invoice was uploaded over 75 days ago and may be subject to penalties in 15 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty <= 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) >= 75
  and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '10 days');

insert into task(claim_id, due_date, task_type, description, insurer, visibility, visibility_role, created_by, created_date, last_modified_by, last_modified_date, version)
select c.id, now() + interval '5 days', 'Invoice Approaching ' || 30*(i.penalty_alert_qty+1) || ' Days', 'The invoice was uploaded over ' || 30*(i.penalty_alert_qty+1)-5 || ' days ago and may be subject to penalties in 5 days time.',
       true, 2, 'ROLE_INS_CH', userId, now(), userId, now(), 0
from claim c, invoice i, insurer ins
where c.invoice_id = i.id
  and i.penalty_alert_qty >= 0 and i.penalty_alert_qty < 2
  and c.insurer_id = ins.id
  and c.claim_type != 3
  and ins.is_task_management_enable = true
  and c.status not in ('InvoicePaymentLogged', 'ClaimClosed', 'InvoiceRejectionAccepted', 'PaymentReceived', 'InvoiceDataCalculationIncorrect')
  and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) >= ((i.penalty_alert_qty+1)*30 - 5)
  and extract(epoch from now() - i.auto_penalty_start)/(3600*24.0) < 90
  and not exists (select * from task where claim_id = c.id and task_type like 'Invoice Approaching%' and now() - created_date < '29 days');


update task set complete = true,
