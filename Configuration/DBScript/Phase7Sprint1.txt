CREATE TABLE penalty_charge
(
  id serial NOT NULL,
  penalty_Name varchar(16) NOT NULL,
  penalty_type varchar(16) NOT NULL,
  penalty_start_age_from integer NOT NULL DEFAULT 0,
  penalty_percentage numeric(10,2) NOT NULL DEFAULT 0.00,
  penalty_percentage_dsc varchar(32) NOT NULL,
  penalty_start_date timestamp without time zone NOT NULL DEFAULT now()
)
