CREATE TABLE penalty_charge
(
  id serial NOT NULL,
  penalty_Name varchar(16),
  penalty_type varchar(16),
  penalty_start_age_from integer DEFAULT 0,
  penalty_start_age_to integer DEFAULT 0,
  penalty_percentage numeric(10,2) NOT NULL DEFAULT 0.00,
  penalty_percentage_dsc character varying(32),
  penalty_start_date timestamp without time zone DEFAULT now()
)
WITH (
  OIDS=FALSE
);
