----------------------------------------------------------------
--POPULATES THE REASON_OF_REJECTION TABLE FOR EACH INSURER
----------------------------------------------------------------
create or replace function populate_desc() returns integer as $$
declare
ins insurer%ROWTYPE;
def_res reason_of_rejection_template%ROWTYPE;
begin 
	for ins in select * from insurer loop
		for def_res in select * from reason_of_rejection_template loop
		 	insert into reason_of_rejection (description , insurer_id , name, type, status, restricted) 
		 	--values (ins.name || ' - ' || def_res.description, ins.id, def_res.name, def_res.type, def_res.status, def_res.restricted);--XXX This is just for testing purposes
	 		values ('', ins.id, def_res.name, def_res.type, def_res.status, def_res.restricted);
	 	end loop;
	 end loop;
	RETURN 1;
END;
$$ LANGUAGE plpgsql;

select * from populate_desc();
drop function populate_desc();

delete from reason_of_rejection where insurer_id is null;
----------------------------------------------------------------
--UPDATES THE EXISTING REFERENCES TO THE REASON_OF_REJECTION_TABLE
----------------------------------------------------------------
UPDATE claim c
SET reason_of_rejection_id = (select id from reason_of_rejection where insurer_id = c.insurer_id 
and name = (select name from reason_of_rejection_template where  id = c.reason_of_rejection_id)
 and type = 'Claim');

UPDATE invoice i
SET reason_of_rejection_id = (select id from reason_of_rejection where insurer_id = (select insurer_id from claim where invoice_id = i.id)
 and name = (select name from reason_of_rejection_template where  id = i.reason_of_rejection_id)
 and type = 'Invoice');

UPDATE audit_trail au
SET invoice_reason_of_rejection = (select id from reason_of_rejection where insurer_id = (select insurer_id from claim where id = au.claim_id)
 and name = (select name from reason_of_rejection_template where  id = au.invoice_reason_of_rejection)
 and type = 'Invoice');
 
UPDATE audit_trail au
SET claim_reason_of_rejection = (select id from reason_of_rejection where insurer_id = (select insurer_id from claim where id = au.claim_id)
 and name = (select name from reason_of_rejection_template where  id = au.invoice_reason_of_rejection)
 and type = 'Claim');
