-----------------------------------------------------
-- BUG#2218 - Production - Audit Trail clean-up   --
-----------------------------------------------------

CREATE OR REPLACE FUNCTION alterAuditTrail()
  RETURNS boolean AS
$BODY$
DECLARE

auditRecord RECORD;
 
BEGIN

FOR auditRecord IN 

select 
    c.cho_reference as cho_reference, a.id as delete_audit, a1.id as update_audit
from 
     audit_trail a, audit_trail a1,  audit_trail a2, claim c
where
     a.claim_id = c.id
     and a1.claim_id = c.id
     and a2.claim_id = c.id
     and a.created_date > a1.created_date
     and a1.created_date > a2.created_date
     and a.reverted = false
     and a1.reverted = false
     and a2.reverted = false
     and a1.new_status = 'ClaimClosed'
     and a1.original_status = a.new_status
     and a1.original_status = a2.new_status
     and a1.new_status = a.original_status
     and not exists (select * from audit_trail a4 where a4.claim_id = c.id and a4.reverted = false and a4.created_date < a.created_date and a4.created_date > a1.created_date)
     and not exists (select * from audit_trail a4 where a4.claim_id = c.id and a4.reverted = false and a4.created_date < a.created_date and a4.created_date < a1.created_date and a4.created_date > a2.created_date)
     order by c.cho_reference

LOOP

RAISE NOTICE 'choref = %  delete_audit_id= % updated_audit_id=  %', auditRecord.cho_reference, auditRecord.delete_audit, auditRecord.update_audit;

UPDATE 
       audit_trail 
SET
    reverted = true,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.update_audit;


DELETE FROM
    audit_trail
WHERE
    id = auditRecord.delete_audit;
  
END LOOP;


RAISE NOTICE 'Updating ClaimClosed to AwaitingInvoicePayment audit_trail entries';

FOR auditRecord IN 

select 
     c.cho_reference as cho_reference, a.id as revert_audit, a.original_status as revert_original_status, a1.id as update_audit
from 
     audit_trail a, audit_trail a1, claim c
where
     a.claim_id = c.id
     and a1.claim_id = c.id
     and a.created_date < a1.created_date
     and a.reverted = false
     and a1.reverted = false
     and a.new_status = 'ClaimClosed'
     and a1.new_status = 'AwaitingInvoicePayment'
     and a1.original_status = a.new_status
     and not exists (select * from audit_trail a2 where a2.claim_id = c.id and a2.reverted = false and a2.created_date < a1.created_date and a2.created_date > a.created_date)

LOOP

 RAISE NOTICE 'choref = %  revert_audit_id= % updated_audit_id=  %', auditRecord.cho_reference, auditRecord.revert_audit, auditRecord.update_audit;

UPDATE 
    audit_trail 
SET
    reverted = true,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.revert_audit;


UPDATE 
    audit_trail 
SET
    original_status = auditRecord.revert_original_status,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.update_audit;


END LOOP;


RAISE NOTICE 'Updating InvoiceRejectionAccepted to AwaitingInvoicePayment audit_trail entries';

FOR auditRecord IN 

select 
     c.cho_reference as cho_reference, a.id as revert_audit, a.original_status as revert_original_status, a1.id as update_audit
from 
     audit_trail a, audit_trail a1, claim c
where
     a.claim_id = c.id
     and a1.claim_id = c.id
     and a.id < a1.id
     and a.reverted = false
     and a1.reverted = false
     and a.new_status = 'InvoiceRejectionAccepted'
     and a1.original_status = a.new_status
     and not exists (select * from audit_trail a2 where a2.claim_id = c.id and a2.reverted = false and a2.id < a1.id and a2.id > a.id)

LOOP

 RAISE NOTICE 'choref = %  revert_audit_id= % updated_audit_id=  %', auditRecord.cho_reference, auditRecord.revert_audit, auditRecord.update_audit;

UPDATE 
    audit_trail 
SET
    reverted = true,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.revert_audit;


UPDATE 
    audit_trail 
SET
    original_status = auditRecord.revert_original_status,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.update_audit;


END LOOP;


RAISE NOTICE 'Updating switch claim to related insurer audit_trail entries';

FOR auditRecord IN 

select 
       c.cho_reference, c.id as claim_id, a.id as first_audit_id_after_switch
from
       claim c, comment cm, audit_trail a
where
       cm.claim_id = c.id
       and cm.reverted = false
       and a.claim_id = c.id
       and a.created_date <= cm.created_date
       and a.reverted = false
       and a.new_status in ('ClaimUnacknowledgedUnrouted','ClaimUnacknowledgedUnassigned','ClaimUnacknowledgedRouted')
       and (cm.comment ilike 'Claim switched from%' or cm.comment ilike 'Claim has been switched from%')
       and not exists (select * from comment cm1 where cm1.claim_id = c.id and (cm1.comment ilike 'Claim switched from%' or cm1.comment ilike 'Claim has been switched from%') and cm1.created_date > cm.created_date and cm1.reverted = false)
       and not exists (select * from audit_trail a1 where a1.claim_id = c.id and a1.created_date <= cm.created_date and a1.id > a.id and a1.reverted = false)
LOOP

 RAISE NOTICE 'choref = %  first audit entry id after switch = % ', auditRecord.cho_reference, auditRecord.first_audit_id_after_switch;

UPDATE 
    audit_trail 
SET
    original_status = '',
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = auditRecord.first_audit_id_after_switch;


UPDATE 
    audit_trail 
SET
    reverted = true,
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    claim_id = auditRecord.claim_id
    and reverted = false
    and id < auditRecord.first_audit_id_after_switch;


END LOOP;

RAISE NOTICE 'Manual update where no entries present in the comment table about switching claims';

UPDATE 
    audit_trail 
SET
    original_status = '',
    last_modified_date = now(),
    last_modified_by = 999
WHERE
    id = 87904;


UPDATE 
    audit_trail 
SET
    reverted = true,
    last_modified_date = now(),
    last_modified_by = 999
FROM audit_trail a
WHERE
    a.id = 87904
    and audit_trail.claim_id = a.claim_id
    and audit_trail.reverted = false
    and audit_trail.id < a.id;

RETURN TRUE;

END;
$BODY$
LANGUAGE plpgsql VOLATILE
  COST 100;

select alterAuditTrail(); 

DROP FUNCTION alterAuditTrail();
