-- View: rpt_claim_audit_trail
--
CREATE OR REPLACE VIEW rpt_claim_audit_trail AS 
 SELECT audit_trail.id, audit_trail.update_date, audit_trail.user_id, audit_trail.claim_id, audit_trail.original_status, audit_trail.new_status, audit_trail.created_by, audit_trail.created_date, audit_trail.last_modified_by, audit_trail.last_modified_date, claim.insurer_id, claim.chorganisation_id, invoice.full_total_to_pay AS total_to_pay, invoice.hire_penalty_charge AS penalty_charge, invoice.id AS invoice_id, 
        CASE
            WHEN audit_trail.claim_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.claim_reason_of_rejection
        END AS claim_reason_of_rejection, 
        CASE
            WHEN audit_trail.invoice_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.invoice_reason_of_rejection
        END AS invoice_reason_of_rejection
   FROM audit_trail audit_trail
   JOIN claim claim ON claim.id = audit_trail.claim_id and audit_trail.reverted = false
   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id;

ALTER TABLE rpt_claim_audit_trail OWNER TO chox;
GRANT ALL ON TABLE rpt_claim_audit_trail TO chox;
GRANT SELECT, UPDATE, INSERT ON TABLE rpt_claim_audit_trail TO chox_user;

--
-- View: vw_claim_monthly_summary
--
CREATE OR REPLACE VIEW vw_claim_monthly_summary AS 
 SELECT sqlgetdayofmonth() AS start_date, date(now()) AS end_date, claim.insurer_id, claim.chorganisation_id, audit.new_status AS status, 0 AS w_count, count(*) AS m_count, 0 AS a_count, 0 AS w_inv_count, count(invoice.id) AS m_inv_count, 0 AS a_inv_count, 0 AS w_total_to_pay, 
        CASE
            WHEN sum(invoice.total_to_pay) IS NULL THEN 0::numeric
            ELSE sum(invoice.total_to_pay)
        END AS m_total_to_pay, 0 AS a_total_to_pay, 0 AS w_penalty_charge, 
        CASE
            WHEN sum(invoice.total_penalty_charge) IS NULL THEN 0::numeric
            ELSE sum(invoice.total_penalty_charge)
        END AS m_penalty_charge, 0 AS a_penalty_charge
   FROM audit_trail audit
   JOIN ( SELECT a.claim_id, max(a.update_date) AS max_update_date
           FROM audit_trail a
          WHERE date(a.update_date) >= sqlgetdayofmonth() AND date(a.update_date) <= date(now()) AND a.reverted = false
          GROUP BY a.claim_id) max_audit ON audit.claim_id = max_audit.claim_id AND audit.update_date = max_audit.max_update_date
   JOIN claim claim ON audit.claim_id = claim.id and audit.reverted = false
   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id
  WHERE NOT (EXISTS ( SELECT a.id, a.update_date, a.user_id, a.claim_id, a.original_status, a.new_status, a.created_by, a.created_date, a.last_modified_by, a.last_modified_date, a.claim_reason_of_rejection, a.invoice_reason_of_rejection, a.version
   FROM audit_trail a
  WHERE a.claim_id = audit.claim_id AND a.reverted = false AND a.update_date = max_audit.max_update_date AND a.id > audit.id))
  GROUP BY claim.insurer_id, claim.chorganisation_id, audit.new_status;

ALTER TABLE vw_claim_monthly_summary OWNER TO chox;
GRANT ALL ON TABLE vw_claim_monthly_summary TO chox;
GRANT SELECT ON TABLE vw_claim_monthly_summary TO chox_user;

--
-- Function: update_user_service(integer)
-- 
CREATE OR REPLACE FUNCTION update_user_service(integer)
  RETURNS void AS
$BODY$
-- First, update stats that are < 5 days old
update user_service
	set outstanding0_5 = (select case when count(*) is null then -1 else count(*) end
from (select case when EXTRACT(DAY FROM (user_service.week_start - aa.update_date)) is null then -1
             else EXTRACT(DAY FROM (user_service.week_start - aa.update_date))
			- COUNT_FULL_WEEKEND_DAYS(aa.update_date::date, user_service.week_start) end as total_day
	from claim cc, audit_trail aa
	where cc.id = aa.claim_id
          and cc.claim_owner_id = user_service.user_id
          and cc.workgroup_id = user_service.workgroup_id
	  and cc.insurer_id = $1 and aa.reverted = false
          and aa.update_date::date < user_service.week_start
          and aa.update_date = (select max(update_date) from audit_trail at where claim_id = cc.id and update_date < user_service.week_start and reverted = false)
	  and aa.new_status in ('ClaimUnacknowledgedRouted', 'ClaimRejectionContested', 'ClaimUpdatedByEngineer', 'InvoiceReferredToClaimsHandler', 'InvoiceEscalatedToHandler', 'ContestedInvoiceReferredToInsurer', 'InvoiceApprovedByBRE', 'AwaitingInvoicePayment')) b
where b.total_day <= 5 and b.total_day >= 0
   ),
            achieved90 = false
where week_start > (current_date - 6) and outstanding > 0;


-- insert user entries
insert into user_service(user_id, workgroup_id, week_start)
select distinct wuw.user_id, wuw.workgroup_id, monday
from web_user wu, web_user_workgroup wuw, week_start, web_user_user_role wuur, web_user_role wur
where wuw.user_id = wuur.web_user_id and wuur.web_user_role_id = wur.id and wur.name = 'ROLE_INS_CH'
  and week_start.monday < current_date
  and wu.id = wuw.user_id
  and wu.status = true
  and wu.insurer_id = $1
  and not exists (select * from user_service where user_service.user_id=wu.id and user_service.workgroup_id=wuw.workgroup_id and user_service.week_start= week_start.monday);

-- Add users that do not belong to workgroups
insert into user_service(user_id, week_start)
select distinct wuur.web_user_id, monday
from week_start, web_user wu, web_user_user_role wuur, web_user_role wur
where wuur.web_user_role_id = wur.id and wur.name = 'ROLE_INS_CH'
  and wu.id = wuur.web_user_id
  and wu.insurer_id = $1
  and week_start.monday < current_date
  and not exists (select * from user_service where user_id = wu.id and user_service.week_start = week_start.monday and workgroup_id is null);

update user_service
	set outstanding = (select count(*) from audit_trail a, claim c
		where c.id = a.claim_id
		   and c.claim_owner_id = user_service.user_id
		   and c.workgroup_id = user_service.workgroup_id
		   and c.insurer_id = $1 and a.reverted = false
		   and a.new_status in ('ClaimUnacknowledgedRouted', 'ClaimRejectionContested', 'ClaimUpdatedByEngineer', 'InvoiceReferredToClaimsHandler', 					'InvoiceEscalatedToHandler','ContestedInvoiceReferredToInsurer', 'InvoiceApprovedByBRE', 'AwaitingInvoicePayment')
		   and a.update_date::date < user_service.week_start
		   and a.update_date = (select max(update_date) from audit_trail at where at.claim_id = c.id and at.reverted = false and at.update_date < user_service.week_start)
   	)
where outstanding is null;

update user_service
	set outstanding0_5 = (select count(*) from (select case when EXTRACT(DAY FROM (user_service.week_start - aa.update_date)) is null then 0
             							else EXTRACT(DAY FROM (user_service.week_start - aa.update_date))
							- COUNT_FULL_WEEKEND_DAYS(aa.update_date::date, user_service.week_start) end as total_day
						    from claim cc, audit_trail aa
						    where cc.id = aa.claim_id
					              and cc.claim_owner_id = user_service.user_id
					              and cc.workgroup_id = user_service.workgroup_id
					              and cc.insurer_id = $1 and aa.reverted = false
					              and aa.update_date::date < user_service.week_start
					              and aa.update_date = (select max(update_date) from audit_trail at where claim_id = cc.id and reverted = false and update_date < user_service.week_start)
					              and aa.new_status in ('ClaimUnacknowledgedRouted', 'ClaimRejectionContested', 'ClaimUpdatedByEngineer', 													    'InvoiceReferredToClaimsHandler', 'InvoiceEscalatedToHandler', 'ContestedInvoiceReferredToInsurer',
									    'InvoiceApprovedByBRE', 'AwaitingInvoicePayment')) b
			      where b.total_day <= 5 and b.total_day >= 0
	)
where outstanding0_5 is null and outstanding > 0;


update user_service
 set outstanding0_5 = 0
where outstanding = 0 and outstanding0_5 is null;

-- Populate achieved90 column
update user_service
   set achieved90 = true
where ((outstanding > 0 and (outstanding0_5*100/outstanding) >= 90) or (outstanding = 0)) and achieved90 = false;

$BODY$
  LANGUAGE 'sql' VOLATILE STRICT
  COST 100;
ALTER FUNCTION update_user_service(integer) OWNER TO chox;
GRANT EXECUTE ON FUNCTION update_user_service(integer) TO chox;
GRANT EXECUTE ON FUNCTION update_user_service(integer) TO public;
GRANT EXECUTE ON FUNCTION update_user_service(integer) TO chox_user;

$BODY$

userId int;
timeMarker timestamp;

userId=$1;

timeMarker=now();

from claim c
where created_date >= SqlGetDayOfWeek()
