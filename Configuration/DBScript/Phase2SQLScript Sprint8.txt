CREATE SEQUENCE workgroup_id_seq
  INCREMENT 1
  MINVALUE 1
  MAXVALUE 9223372036854775807
  START 1
  CACHE 1;
ALTER TABLE workgroup_id_seq OWNER TO postgres;

CREATE TABLE workgroup
(
  id integer NOT NULL DEFAULT nextval('workgroup_id_seq'::regclass),
  name character varying NOT NULL,
  insurer_id integer NOT NULL,
  status boolean NOT NULL DEFAULT false,
  created_by integer NOT NULL,
  created_date timestamp without time zone NOT NULL DEFAULT now(),
  last_modified_by integer NOT NULL,
  last_modified_date timestamp without time zone NOT NULL DEFAULT now(),
  CONSTRAINT workgroup_pkey PRIMARY KEY (id),
  CONSTRAINT workgroup_insurer_fkey FOREIGN KEY (insurer_id)
      REFERENCES insurer (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (OIDS=FALSE);
ALTER TABLE workgroup OWNER TO postgres;

-- -------------------------------------------------------------------------------------------

CREATE SEQUENCE user_workgroup_id_seq
  INCREMENT 1
  MINVALUE 1
  MAXVALUE 9223372036854775807
  START 1
  CACHE 1;
ALTER TABLE user_workgroup_id_seq OWNER TO postgres;

CREATE TABLE user_workgroup
(
  id integer NOT NULL DEFAULT nextval('user_workgroup_id_seq'::regclass),
  user_id integer NOT NULL,
  workgroup_id integer NOT NULL,
  created_by integer NOT NULL,
  created_date timestamp without time zone NOT NULL DEFAULT now(),
  last_modified_by integer NOT NULL,
  last_modified_date timestamp without time zone NOT NULL DEFAULT now(),
  CONSTRAINT user_workgroup_pkey PRIMARY KEY (id),
  CONSTRAINT userworkgroup_web_user_fkey FOREIGN KEY (user_id) REFERENCES web_user (id) MATCH SIMPLE ON UPDATE NO ACTION ON DELETE NO ACTION,
  CONSTRAINT userworkgroup_workgroup_fkey FOREIGN KEY (workgroup_id) REFERENCES workgroup (id) MATCH SIMPLE ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (OIDS=FALSE);
ALTER TABLE workgroup OWNER TO postgres;
