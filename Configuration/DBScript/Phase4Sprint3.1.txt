CREATE TABLE task (
  id serial NOT NULL,
  claim_id integer,
  due_date timestamp without time zone,
  completed_date timestamp without time zone,
  task_type character varying(32),
  description character varying(256),
  complete boolean not null default false,
  insurer boolean not null,
  visibility smallint,
  visibility_role character varying(16),
  related_task integer,
  created_by integer,
  completed_by integer,
  last_modified_by integer,
  last_modified_date timestamp without time zone,
  created_date timestamp without time zone,
  version integer,
  CONSTRAINT task_pkey PRIMARY KEY (id),
  CONSTRAINT task_claim_fkey FOREIGN KEY (claim_id)
      REFERENCES claim (id) MATCH SIMPLE
      ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (OIDS=FALSE);
grant select, update, insert on task to chox_user;
grant select, update on task_id_seq to chox_user;
