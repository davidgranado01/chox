ALTER TABLE insurer ADD COLUMN is_upload_enabled boolean not null default false;
ALTER TABLE chorganisation ADD COLUMN insurer_upload_only boolean not null default false;
ALTER TABLE claim ADD COLUMN insurer_upload boolean not null default false;

CREATE TABLE chorganisation_alias (
	id serial NOT NULL,
	chorganisation_id integer not null,
	alias_name character varying(16) not null,
	created_by integer,
  	last_modified_by integer,
  	last_modified_date timestamp without time zone,
  	created_date timestamp without time zone,
  	version integer,
  	CONSTRAINT chorganisation_alias_pkey PRIMARY KEY (id),
	CONSTRAINT chorganisation_id_fkey FOREIGN KEY (chorganisation_id)
		REFERENCES chorganisation (id) MATCH SIMPLE ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT created_by_fkey FOREIGN KEY (created_by)
		REFERENCES web_user (id) MATCH SIMPLE ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT last_modified_by_fkey FOREIGN KEY (last_modified_by)
		REFERENCES web_user (id) MATCH SIMPLE ON UPDATE NO ACTION ON DELETE NO ACTION
)
WITH (OIDS=FALSE);
grant select, update, insert on chorganisation_alias to chox_user;
grant select, update on chorganisation_alias_id_seq to chox_user;
