insert into web_user_role(version, name, description, type_id, is_workgroup_related, is_ownership_related,
                          created_by, created_date, last_modified_by, last_modified_date)
select 0, 'ROLE_CHO_USER', 'User Manager', 3, false, false, 999, now(), 999, now();

update accessibility_item set role='ROLE_CHO_USER'
from accessibility
where accessibility_item.accessibility_id = accessibility.id
  and accessibility.name = 'menu.Admin'
  and accessibility_item.role = 'ROLE_CHO_MNG';


update accessibility_item set role='ROLE_CHO_USER'
from accessibility
where accessibility_item.accessibility_id = accessibility.id
  and accessibility.name = 'admin.UserManagement'
  and accessibility_item.role = 'ROLE_CHO_MNG';
  


delete from 
      accessibility_item 
where 
      role = 'ROLE_CHO' 
      and accessibility_id in (select id from accessibility where name = 'menu.Inbox');

delete from 
      accessibility_item 
where 
      role = 'ROLE_CHO' 
      and accessibility_id in (select id from accessibility where name = 'menu.Search');

insert into accessibility_item(role, access_right, accessibility_id)
select 'ROLE_CHO_MNG', 1, id from accessibility where name='menu.Search';

insert into accessibility_item(role, access_right, accessibility_id)
select 'ROLE_CHO_MNG', 1, id from accessibility where name='menu.Inbox';


insert into accessibility_item(role, access_right, accessibility_id)
select 'ROLE_CHO_OPR', 1, id from accessibility where name='menu.Search';

insert into accessibility_item(role, access_right, accessibility_id)

insert into accessibility_item(role, access_right, accessibility_id)
select 'ROLE_CHO_MI', 1, id from accessibility where name='menu.Search';

insert into accessibility_item(role, access_right, accessibility_id)
select 'ROLE_CHO_MI', 1, id from accessibility where name='menu.Inbox';
