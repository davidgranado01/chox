 SELECT audit_trail.id, audit_trail.update_date, audit_trail.user_id, audit_trail.claim_id, audit_trail.original_status, 
 audit_trail.new_status, audit_trail.created_by, audit_trail.created_date, audit_trail.last_modified_by, audit_trail.last_modified_date, 
 claim.insurer_id, claim.chorganisation_id, invoice.total_to_pay, invoice.penalty_charge AS penalty_charge, invoice.id AS invoice_id, 
        CASE
            WHEN audit_trail.claim_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.claim_reason_of_rejection
        END AS claim_reason_of_rejection, 
        CASE
            WHEN audit_trail.invoice_reason_of_rejection = 0 THEN NULL::integer
            ELSE audit_trail.invoice_reason_of_rejection
        END AS invoice_reason_of_rejection
   FROM audit_trail audit_trail
   JOIN claim claim ON claim.id = audit_trail.claim_id
   LEFT JOIN invoice invoice ON claim.invoice_id = invoice.id;
