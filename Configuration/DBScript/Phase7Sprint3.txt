-- 7.3.1 Refactoring of functional access to take into account the claim type      --
-------------------------------------------------------------------------------------
ALTER TABLE accessibility ADD COLUMN claim_type integer;
CREATE UNIQUE INDEX accessibility_ui ON accessibility(name, claim_type);

UPDATE accessibility set name = 'activity' || substring(name, 7)
WHERE name like 'button.%';

UPDATE accessibility set name = 'activity.RevertClaim' || substring(name, 27)
WHERE name like 'activity.RevertClaimStatus.%';

UPDATE accessibility set name = 'activity.PaymentNotReceived.InvoicePaymentLogged' WHERE name like 'activity.UpdatePaymentNotReceived.InvoicePaymentLogged';

UPDATE accessibility set name = 'activity' || substring(name, 7)
WHERE name like 'action.%';

UPDATE accessibility set name = 'activity.AssignWorkgroup' || substring(name, 35)
WHERE name like 'activity.updateclaimlineofbusiness.%';

UPDATE accessibility set name = 'activity.AcknowledgeClaim' || substring(name, 34)
WHERE name like 'activity.updateclaiminsurerdetail.%';

UPDATE accessibility set name = 'activity.ClaimRejectionContest' || substring(name, 38)
WHERE name like 'activity.contestOrAcceptRejectedClaim.%';

UPDATE accessibility set name = 'activity.AcknowledgeClaim.ClaimRejectionContested'
WHERE name = 'activity.approveContestedClaim.ClaimRejectionContested';

UPDATE accessibility set name = 'activity.ClaimAwaitingCarHireInfo' || substring(name, 34)
WHERE name like 'activity.updateclaimcarhiredetail.%';

UPDATE accessibility set name = 'activity.InvoiceResubmit' || substring(name, 51)
WHERE name like 'activity.updateinvoicedtailforcalculationincorrect.%';

UPDATE accessibility set name = 'activity.InvoiceAccepted' || substring(name, 31)
WHERE name like 'activity.approveBREPassedClaim.%';

UPDATE accessibility set name = 'activity.InvoicePaymentLogged' || substring(name, 27)
WHERE name like 'activity.logInvoicePayment.%';

UPDATE accessibility set name = 'activity.ClaimRegisterByFnol' || substring(name, 31)
WHERE name like 'activity.registeredclaimbyfnol.%';

UPDATE accessibility set name = 'activity.ClaimPending' || substring(name, 22)
WHERE name like 'activity.claimPending.%';

UPDATE accessibility set name = 'activity.InvoicePaymentReceived' || substring(name, 31)
WHERE name like 'activity.updatePaymentReceived.%';

UPDATE accessibility set name = 'activity.AssignOwner' || substring(name, 30)
WHERE name like 'activity.assignClaimOwnership.%';

UPDATE accessibility set name = 'activity.ResolveLiability' || substring(name, 25)
WHERE name like 'activity.updateLiability.%';

UPDATE accessibility set name = 'activity.UpdateManualInvoicePaid' || substring(name, 29)
WHERE name like 'activity.updateManualInvoice.%';

UPDATE accessibility set name = 'activity.AssignManualInvoiceOwner' || substring(name, 46)
WHERE name like 'activity.updateManualInvoiceWorkgroupAndOwner.%';

UPDATE accessibility SET name = 'panel.FNOLReviewed.ClaimUnacknowledgedRouted',
           check_fnol_enabled = true
WHERE name = 'panel.FNOLReviewed';

INSERT INTO accessibility(name, is_workgroup_check, is_ownership_check,
                          check_manual_inv_workgroup_enabled, check_manual_inv_claimownership_enabled,
                          check_fnol_enabled)
SELECT 'panel.FNOLReviewed.ClaimUnacknowledgedUnassigned', false, false, false, false, true;

INSERT INTO accessibility_item(accessibility_id, role, access_right)
	SELECT id, 'ROLE_CHOX_ADMIN', 1
	FROM accessibility
	WHERE name = 'panel.FNOLReviewed.ClaimUnacknowledgedRouted';

INSERT INTO accessibility_item(accessibility_id, role, access_right)
	SELECT a2.id, ai.role, ai.access_right
	FROM accessibility a, accessibility_item ai, accessibility a2
	WHERE a.name = 'panel.FNOLReviewed.ClaimUnacknowledgedRouted'
	  AND ai.accessibility_id = a.id
	  AND a2.name='panel.FNOLReviewed.ClaimUnacknowledgedUnassigned';

-- Delete all accessibility items with access right of 0:
--   this table gives access, not removes
DELETE FROM accessibility_item WHERE access_right=0;
