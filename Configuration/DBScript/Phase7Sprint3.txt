-- 7.3.1 Refactoring of functional access to take into account the claim type      --
-------------------------------------------------------------------------------------
ALTER TABLE accessibility ADD COLUMN claim_type integer;
CREATE UNIQUE INDEX accessibility_ui ON accessibility(name, claim_type);

UPDATE accessibility set name = 'activity' || substring(name, 7)
WHERE name like 'button.%';
