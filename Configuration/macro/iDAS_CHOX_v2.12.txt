Private Function RemoveIllegals(inpStr As String) As String
Dim f As Long
Dim tmpStr As String
Dim opStr As String
tmpStr = Trim(inpStr)
opStr = ""
For f = 1 To Len(tmpStr)

    DoEvents
    
    If (Asc(Mid(tmpStr, f, 1)) > 31 And Asc(Mid(tmpStr, f, 1)) < 127) Or Asc(Mid(tmpStr, f, 1)) = 128 Then
        opStr = opStr & Mid(tmpStr, f, 1)
    End If
    
Next f

RemoveIllegals = opStr
End Function

Private Function ReturnDate(tmpData As String) As String
ReturnDate = CStr(Format(tmpData, "yyyy-mm-ddThh:mm:ss"))

If ReturnDate = "00/01/1900 00:00:00" Or ReturnDate = "1899-12-30T00:00:00" Then
    ReturnDate = ""
End If
End Function
